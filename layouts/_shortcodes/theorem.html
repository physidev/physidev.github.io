{{- /* Renders a boxed theorem/lemma/definition/etc. environment. @param {string} [type] The environment label, e.g. "Theorem", "Lemma", "Definition". Defaults to "Theorem". @param {string} [name] An optional name for the result, e.g. "Clausius' theorem". The first word of `type` picks the box color: Definition or Algorithm; Theorem, Lemma, Proposition, Corollary or Conjecture; Example; anything else (Remark, Notation, ...). Numbering is off by default. To turn it on for a page, set `numbered_theorems = true` in that page's front matter; environments are then numbered per `type` in document order. @example: {{< theorem type="Lemma" name="Abel's lemma" >}} ... {{< /theorem >}} */}} {{- $type := .Get "type" | default "Theorem" -}} {{- $name := .Get "name" -}} {{- $number := "" -}} {{- if .Page.Param "numbered_theorems" -}} {{- $counters := .Page.Store.Get "theoremCounters" | default (dict) -}} {{- $prev := index $counters $type | default 0 -}} {{- $current := add $prev 1 -}} {{- $counters = merge $counters (dict $type $current) -}} {{- .Page.Store.Set "theoremCounters" $counters -}} {{- $number = $current -}} {{- end -}} {{- $label := $type -}} {{- with $number }}{{ $label = printf "%s %v" $label . }}{{ end -}} {{- with $name }}{{ $label = printf "%s (%s)" $label . }}{{ end -}} {{- /* Color group: definitions, claims (theorems and conjectures), examples, everything else. */ -}} {{- $word := index (split $type " ") 0 | lower -}} {{- $kind := "other" -}} {{- if in (slice "definition" "algorithm") $word }}{{ $kind = "definition" -}} {{- else if in (slice "theorem" "lemma" "proposition" "corollary" "conjecture") $word }}{{ $kind = "result" -}} {{- else if eq $word "example" }}{{ $kind = "example" -}} {{- end -}}
{{ .Page.RenderString (printf "**%s.** %s" $label .Inner) }}
